  <action>
    Creare web-ui/lib/autocomplete.ts:
    ```ts
    import type { CatalogStep } from './types';
    export const GHERKIN_PREFIX_RE = /(?:^|\n)(Given|When|Then|And|But)\s+([^\n]*)$/;
    export function getAutocompletePrefix(text: string): string | null { /* match su ultima riga; null se nessuna keyword */ }
    export function getSuggestions(text: string, steps: CatalogStep[]): CatalogStep[] {
      const prefix = getAutocompletePrefix(text);
      if (prefix === null) return [];
      const p = prefix.toLowerCase();
      return steps.filter(s => s.expression.toLowerCase().startsWith(p)).slice(0, 8);
    }
    ```
    Implementare getAutocompletePrefix usando GHERKIN_PREFIX_RE su `text` (il regex con `$` e senza flag multiline ancora il match alla fine della stringa = ultima riga).
