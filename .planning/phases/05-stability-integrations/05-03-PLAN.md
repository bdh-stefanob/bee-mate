```typescript
import { NextResponse } from 'next/server';

interface PushBody {
  content: string;
  filePath: string;
}

export async function POST(request: Request) {
  // 1. Leggere i token dagli header (mai dal body)
  const githubToken  = request.headers.get('x-github-token');
  const githubOwner  = request.headers.get('x-github-owner');
  const githubRepo   = request.headers.get('x-github-repo');
  const githubBranch = request.headers.get('x-github-branch') ?? 'main';

  if (!githubToken || !githubOwner || !githubRepo) {
    return NextResponse.json(
      { ok: false, error: 'Missing required headers: x-github-token, x-github-owner, x-github-repo' },
      { status: 400 }
    );
  }

  // 2. Leggere il body
  let body: PushBody;
  try {
    body = await request.json() as PushBody;
  } catch {
    return NextResponse.json({ ok: false, error: 'Invalid JSON body' }, { status: 400 });
  }

  const { content, filePath } = body;
  if (!content || !filePath) {
    return NextResponse.json(
      { ok: false, error: 'Missing required fields: content, filePath' },
      { status: 400 }
    );
  }

  // 3. Validare filePath (solo caratteri sicuri, nessun path traversal)
  // Accetta: alfanumerici, /, ., -, _ — blocca: .., spazi, caratteri speciali
  if (!/^[\w./-]+$/.test(filePath) || filePath.includes('..')) {
    return NextResponse.json(
      { ok: false, error: 'Invalid filePath' },
      { status: 400 }
    );
  }

  const apiBase = `https://api.github.com/repos/${githubOwner}/${githubRepo}/contents/${filePath}`;
  const authHeader = `token ${githubToken}`;
  const ghHeaders = {
    'Authorization': authHeader,
    'Accept': 'application/vnd.github+json',
    'X-GitHub-Api-Version': '2022-11-28',
    'Content-Type': 'application/json',
  };

  try {
    // 4. Leggere il SHA se il file esiste già (GET)
    let sha: string | undefined;
    const getRes = await fetch(`${apiBase}?ref=${githubBranch}`, {
      headers: ghHeaders,
    });
    if (getRes.ok) {
      const existing = await getRes.json() as { sha: string };
      sha = existing.sha;
    } else if (getRes.status !== 404) {
      // 404 = file non esiste ancora (OK), altri errori = problema reale
      const errData = await getRes.json() as { message?: string };
      throw new Error(errData.message ?? `GitHub GET failed: ${getRes.status}`);
    }

    // 5. Codificare il contenuto in base64 (compatibile Unicode)
    const base64Content = Buffer.from(content, 'utf-8').toString('base64');

    // 6. PUT per creare o aggiornare il file
    const putBody: Record<string, string> = {
      message: `feat: update ${filePath} via BDD web editor`,
      content: base64Content,
      branch: githubBranch,
    };
    if (sha) putBody.sha = sha;

    const putRes = await fetch(apiBase, {
      method: 'PUT',
      headers: ghHeaders,
      body: JSON.stringify(putBody),
    });

    if (!putRes.ok) {
      const errData = await putRes.json() as { message?: string };
      throw new Error(errData.message ?? `GitHub PUT failed: ${putRes.status}`);
    }

    return NextResponse.json({ ok: true });

  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : 'Unknown error';
    // Non esporre il token nell'errore
    const safeMessage = message.replace(githubToken, '[TOKEN]');
    return NextResponse.json({ ok: false, error: safeMessage }, { status: 500 });
  }
}
```

Nota sicurezza:
- Il token viene letto dall'header, non dal body
- filePath validato con regex whitelist prima dell'uso nell'URL
- Il token è rimosso dal messaggio di errore (replace con '[TOKEN]')
- Buffer.from(content, 'utf-8').toString('base64') gestisce Unicode correttamente

```typescript
const { settings } = useSettings();
const [isCommitting, setIsCommitting] = useState(false);

const handleCommitGitHub = useCallback(async () => {
  if (!settings.githubToken || !settings.githubOwner || !settings.githubRepo) return;

  // Helper slugify inline (no dipendenza esterna necessaria)
  const slugify = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');

  // Deriva il filePath dal nome Feature nel contenuto, con fallback
  const featureMatch = content.match(/Feature:\s*(.+)/i);
  const featureName = featureMatch ? featureMatch[1].trim() : 'scenario';
  const slug = slugify(featureName) || 'scenario';
  const filePath = `src/features/${slug}.feature`;

  setIsCommitting(true);
  try {
    const res = await fetch('/api/github/push', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-github-token': settings.githubToken,
        'x-github-owner': settings.githubOwner,
        'x-github-repo': settings.githubRepo,
        'x-github-branch': settings.githubBranch || 'main',
      },
      body: JSON.stringify({ content, filePath }),
    });
    const data = await res.json() as { ok: boolean; error?: string };
    if (!data.ok) throw new Error(data.error ?? 'Unknown error');
    toast.success(t.editor.commitGitHubSuccess);
  } catch (err: unknown) {
    const msg = err instanceof Error ? err.message : 'Unknown error';
    toast.error(`${t.editor.commitGitHubError}: ${msg}`);
  } finally {
    setIsCommitting(false);
  }
}, [content, settings, t, slugify]);
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → POST /api/github/push | Header x-github-token, body {content, filePath} |
| /api/github/push → api.github.com | PUT con Authorization token — server-to-server |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-03-01 | Tampering | filePath in PUT URL | mitigate | Whitelist regex `^[\w./-]+$` + controllo `..` — blocca path traversal nel repo |
| T-05-03-02 | Information Disclosure | GitHub token in error messages | mitigate | `message.replace(githubToken, '[TOKEN]')` prima di restituire l'errore al client |
| T-05-03-03 | Information Disclosure | GitHub token nel body | mitigate | Token letto da `request.headers.get('x-github-token')` — accettato solo come header, mai dal body |
| T-05-03-04 | Spoofing | GitHub API endpoint | accept | Uso localhost — nessun redirect malevolo realistico. Per produzione aggiungere allowlist endpoint. |
| T-05-03-05 | Denial of Service | commit loop | accept | Bottone `disabled` durante isCommitting — nessun debounce necessario per uso locale |
</threat_model>
